# --- Database Models ---

class User(UserMixin, db.Model):
    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(80), unique=True, nullable=False)
    password_hash = db.Column(db.String(250), nullable=False)
    workouts = db.relationship('Workout', backref='owner', lazy=True)

    def set_password(self, password):
        self.password_hash = generate_password_hash(password)

    def check_password(self, password):
        return check_password_hash(self.password_hash, password)

class Workout(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    exercise = db.Column(db.String(100), nullable=False)
    duration_mins = db.Column(db.Integer, nullable=False)
    calories_burned = db.Column(db.Integer, nullable=False)
    date = db.Column(db.Date, nullable=False, default=datetime.utcnow().date)
    user_id = db.Column(db.Integer, db.ForeignKey('user.id'), nullable=False)

@login_manager.user_loader
def load_user(user_id):
    return User.query.get(int(user_id))

# --- Application Routes ---

@app.route("/")
@login_required
def dashboard():
    user_workouts = Workout.query.filter_by(user_id=current_user.id).order_by(Workout.date.desc()).all()
    
    total_workouts = len(user_workouts)
    total_duration = sum(w.duration_mins for w in user_workouts)
    total_calories = sum(w.calories_burned for w in user_workouts)

    return render_template_string(
        HTML_TEMPLATE,
        page="dashboard",
        workouts=user_workouts,
        total_workouts=total_workouts,
        total_duration=total_duration,
        total_calories=total_calories
    )

@app.route("/register", methods=["GET", "POST"])
def register():
    if current_user.is_authenticated:
        return redirect(url_for("dashboard"))
        
    if request.method == "POST":
        username = request.form.get("username")
        password = request.form.get("password")

        if User.query.filter_by(username=username).first():
            flash("Username already exists.")
            return redirect(url_for("register"))

        new_user = User(username=username)
        new_user.set_password(password)
        db.session.add(new_user)
        db.session.commit()
        
        login_user(new_user)
        return redirect(url_for("dashboard"))

    return render_template_string(HTML_TEMPLATE, page="register")

@app.route("/login", methods=["GET", "POST"])
def login():
    if current_user.is_authenticated:
        return redirect(url_for("dashboard"))

    if request.method == "POST":
        username = request.form.get("username")
        password = request.form.get("password")
        user = User.query.filter_by(username=username).first()

        if user and user.check_password(password):
            login_user(user)
            return redirect(url_for("dashboard"))
            
        flash("Invalid username or password.")

    return render_template_string(HTML_TEMPLATE, page="login")

@app.route("/logout")
@login_required
def logout():
    logout_user()
    return redirect(url_for("login"))

@app.route("/add", methods=["POST"])
@login_required
def add_workout():
    exercise = request.form.get("exercise")
    duration = request.form.get("duration")
    calories = request.form.get("calories")

    if exercise and duration and calories:
        workout = Workout(
            exercise=exercise,
            duration_mins=int(duration),
            calories_burned=int(calories),
            user_id=current_user.id
        )
        db.session.add(workout)
        db.session.commit()

    return redirect(url_for("dashboard"))

@app.route("/delete/<int:id>")
@login_required
def delete_workout(id):
    workout = Workout.query.get_or_404(id)
    if workout.user_id == current_user.id:
        db.session.delete(workout)
        db.session.commit()
    return redirect(url_for("dashboard"))

# --- JSON API Route ---

@app.route("/api/workouts", methods=["GET"])
@login_required
def get_workouts_api():
    user_workouts = Workout.query.filter_by(user_id=current_user.id).all()
    data = [
        {
            "id": w.id,
            "exercise": w.exercise,
            "duration_mins": w.duration_mins,
            "calories_burned": w.calories_burned,

        }

    return jsonify({"workouts": data})
